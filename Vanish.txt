-- ============================================================
-- Remote limiter (token bucket on FireServer)
-- ============================================================

local bucket = { tokens = state.BurstLimit, last = os.clock() }
local isOwnCall = checkcaller or function() return false end

local function shouldAllow(now)
    if not state.Enabled then return true end

    local elapsed = now - bucket.last
    bucket.last = now
    bucket.tokens = math.min(state.BurstLimit, bucket.tokens + elapsed * state.MaxPPS)

    if bucket.tokens >= 1 then
        bucket.tokens -= 1
        return true
    end

    state.totalDropped += 1
    if state.Log then
        warn("[Vanish] Remote call limited")
    end
    return false
end

local oldNamecall
local function hookFn(self, ...)
    if state.Enabled and not state.unloaded
        and getnamecallmethod() == "FireServer"
        and not isOwnCall()
        and not shouldAllow(os.clock()) then
        return
    end
    return oldNamecall(self, ...)
end

state.hookOk = pcall(function()
    local wrap = newcclosure or function(f) return f end
    if hookmetamethod then
        oldNamecall = hookmetamethod(game, "__namecall", wrap(hookFn))
    else
        local mt = getrawmetatable(game)
        setreadonly(mt, false)
        oldNamecall = mt.__namecall
        mt.__namecall = wrap(hookFn)
        setreadonly(mt, true)
    end
end)

if not state.hookOk or not oldNamecall then
    state.hookOk = false
    warn("[Vanish] Remote hook failed - limiter off, lag detector still running")
end

local tracks = {}
local selfLagUntil = 0
local lastBeat = os.clock()

local pingItem = nil
local nextPingLookup = 0
local function readPing(now)
    if not pingItem then
        if now < nextPingLookup then return 0 end
        nextPingLookup = now + 2
        local ok, item = pcall(function()
            return Stats.Network.ServerStatsItem["Data Ping"]
        end)
        if ok and item then
            pingItem = item
        else
            return 0
        end
    end
    local ok, v = pcall(pingItem.GetValue, pingItem)
    return (ok and tonumber(v)) or 0
end

local function setHighlight(t, on)
    if on then
        if (not t.hl or not t.hl.Parent) and t.char then
            local hl = Instance.new("Highlight")
            hl.Name = "VanishLagFlag"
            hl.FillColor = Color3.fromRGB(255, 60, 60)
            hl.FillTransparency = 0.6
            hl.OutlineColor = Color3.fromRGB(255, 60, 60)
            hl.Parent = t.char
            t.hl = hl
        end
    elseif t.hl then
        t.hl:Destroy()
        t.hl = nil
    end
end

local function clearTrack(plr)
    local t = tracks[plr]
    if t then
        setHighlight(t, false)
        tracks[plr] = nil
    end
end

local playerList = {}
local function rebuildPlayers()
    table.clear(playerList)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            playerList[#playerList + 1] = p
        end
    end
end
rebuildPlayers()

connect(Players.PlayerAdded, function()
    rebuildPlayers()
end)
connect(Players.PlayerRemoving, function(plr)
    clearTrack(plr)
    task.defer(rebuildPlayers)
end)

connect(RunService.Heartbeat, function()
    if state.unloaded then return end

    local now = os.clock()
    local dt = now - lastBeat
    lastBeat = now

    local ping = readPing(now)
    state.ping = ping

    if dt > 0.2 or ping > 450 then
        selfLagUntil = now + 1.5
    end
    local selfLagging = now < selfLagUntil
    state.selfLagging = selfLagging

    local flaggedList = state.flagged
    table.clear(flaggedList)

    for i = 1, #playerList do
        local plr = playerList[i]
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")

        if not hrp then
            clearTrack(plr)
        else
            local t = tracks[plr]
            if not t or t.char ~= char then
                if t then setHighlight(t, false) end
                t = {
                    char = char,
                    pos = hrp.Position,
                    lastChange = now,
                    lastVel = 0,
                    flagUntil = 0,
                    spikes = 0,
                    status = "OK",
                    wasFlagged = false,
                    lastToast = -100,
                }
                tracks[plr] = t
            end

            local pos = hrp.Position
            local delta = (pos - t.pos).Magnitude

            if selfLagging then
                t.pos = pos
                t.lastChange = now
                t.lastVel = 0
            elseif delta > 0.05 then
                local gap = now - t.lastChange
                if gap >= DETECT.FreezeTime
                    and t.lastVel >= DETECT.MinSpeed
                    and delta >= DETECT.JumpStuds then
                    t.spikes += 1
                    t.flagUntil = now + DETECT.FlagHold
                    if state.Log then
                        warn(string.format("[Vanish] %s lag spike: froze %.2fs, snapped %.0f studs", plr.Name, gap, delta))
                    end
                end
                t.pos = pos
                t.lastChange = now
                t.lastVel = hrp.AssemblyLinearVelocity.Magnitude
            end

            local frozen = (not selfLagging)
                and (now - t.lastChange) >= DETECT.FreezeTime
                and t.lastVel >= DETECT.MinSpeed

            local status = "OK"
            if now < t.flagUntil then
                status = "SPIKE"
            elseif frozen then
                status = "FROZEN"
            end
            t.status = status
            local flagged = status ~= "OK"

            setHighlight(t, flagged and state.Highlight)

            if flagged and not t.wasFlagged and (now - t.lastToast) > 8 then
                t.lastToast = now
                if state.onFlag then
                    pcall(state.onFlag, plr, t)
                end
            end
            t.wasFlagged = flagged

            if flagged then
                flaggedList[#flaggedList + 1] = plr
            end
        end
    end
end)

local infoCache = {}
local function tweenInfo(t, style)
    style = style or Enum.EasingStyle.Quad
    local key = tostring(t) .. style.Name
    local info = infoCache[key]
    if not info then
        info = TweenInfo.new(t, style, Enum.EasingDirection.Out)
        infoCache[key] = info
    end
    return info
end

local function tween(o, t, props, style)
    local tw = TweenService:Create(o, tweenInfo(t, style), props)
    tw:Play()
    return tw
end

-- One shared drag handler for the window, the pill and every slider.
local activeDrag = nil
local function isPointerStart(input)
    return input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch
end

connect(UserInputService.InputChanged, function(input)
    if activeDrag and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
        activeDrag(input.Position)
    end
end)

connect(UserInputService.InputEnded, function(input)
    if activeDrag and isPointerStart(input) then
        activeDrag = nil
    end
end)

local function viewport()
    local cam = workspace.CurrentCamera
    return cam and cam.ViewportSize or Vector2.new(1280, 720)
end

local WIN_W, WIN_H = 372, 468

-- Phone or tablet without a keyboard counts as mobile.
local isMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled

local function effectiveScale()
    if not config.autoScale then
        return config.scale / 100
    end
    local vp = viewport()
    local base = isMobile and 0.8 or 1
    local fit = math.min((vp.X - 24) / WIN_W, (vp.Y - 24) / WIN_H)
    return math.clamp(math.min(base, fit), 0.5, 1)
end

local function makeDraggable(handle, target, canDrag, clampFn)
    local moved = false
    handle.InputBegan:Connect(function(input)
        if not isPointerStart(input) then return end
        if canDrag and not canDrag() then return end
        moved = false
        local dragStart = input.Position
        local startPos = target.Position
        activeDrag = function(pos)
            local d = pos - dragStart
            if math.abs(d.X) > 6 or math.abs(d.Y) > 6 then
                moved = true
            end
            local x = startPos.X.Offset + d.X
            local y = startPos.Y.Offset + d.Y
            if clampFn then
                x, y = clampFn(x, y)
            end
            target.Position = UDim2.new(startPos.X.Scale, x, startPos.Y.Scale, y)
        end
    end)
    return function()
        return moved
    end
end

local function createSlider(parent, name, min, max, default, fmt, callback)
    local holder = card(parent, 48)
    label(holder, { Size = UDim2.new(0.6, 0, 0, 18), Position = UDim2.new(0, 12, 0, 6), Text = name })
    local valueLbl = label(holder, {
        Size = UDim2.new(0.4, -12, 0, 18),
        Position = UDim2.new(0.6, 0, 0, 6),
        TextXAlignment = Enum.TextXAlignment.Right,
        Font = Enum.Font.GothamBold,
    })

    local hit = make("TextButton", {
        Size = UDim2.new(1, -24, 0, 20),
        Position = UDim2.new(0, 12, 0, 24),
        BackgroundTransparency = 1,
        Text = "",
        AutoButtonColor = false,
    }, holder)

    local bar = make("Frame", {
        Size = UDim2.new(1, 0, 0, 6),
        Position = UDim2.new(0, 0, 0.5, -3),
        BackgroundColor3 = Color3.fromRGB(46, 46, 58),
        BorderSizePixel = 0,
    }, hit)
    corner(bar, 3)

    local fill = make("Frame", { Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0 }, bar)
    corner(fill, 3)
    bindAccent(fill, "BackgroundColor3")

    local knob = make("Frame", {
        Size = UDim2.new(0, 14, 0, 14),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0),
        BackgroundColor3 = WHITE,
        BorderSizePixel = 0,
        ZIndex = 2,
    }, bar)
    corner(knob, 7)

    local current = nil
    local function show(value)
        local rel = (value - min) / (max - min)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, 0, 0.5, 0)
        valueLbl.Text = fmt(value)
    end
    show(default)
    current = default

    local function update(x)
        local w = math.max(bar.AbsoluteSize.X, 1)
        local rel = math.clamp((x - bar.AbsolutePosition.X) / w, 0, 1)
        local value = math.floor(min + (max - min) * rel + 0.5)
        if value == current then return end
        current = value
        show(value)
        callback(value)
    end

    hit.InputBegan:Connect(function(input)
        if isPointerStart(input) then
            activeDrag = function(pos)
                update(pos.X)
            end

-- Updaters only run for the visible tab (tab 0 = always).
local updaters = {}
local currentTab = 1
local menuOpen = true

local function addUpdater(tab, fn)
    updaters[#updaters + 1] = { tab = tab, fn = fn }
end

local function runUpdaters()
    for _, u in ipairs(updaters) do
        if u.tab == 0 or (menuOpen and u.tab == currentTab) then
            local ok, err = pcall(u.fn)
            if not ok and state.Log then
                warn("[Vanish] UI update error: " .. tostring(err))
            end
        end
    end
end

    local rows = {}

    local function createRow(plr)
        local f = make("Frame", {
            Size = UDim2.new(1, 0, 0, 28),
            BackgroundColor3 = Color3.fromRGB(38, 38, 50),
            BackgroundTransparency = 0.35,
            BorderSizePixel = 0,
        }, listFrame)
        corner(f, 7)
        local n = label(f, {
            Size = UDim2.new(1, -92, 1, 0),
            Position = UDim2.new(0, 10, 0, 0),
            Text = plr.DisplayName,
            TextSize = 11,
            TextTruncate = Enum.TextTruncate.AtEnd,
        })
        local chip = label(f, {
            Size = UDim2.new(0, 76, 0, 20),
            Position = UDim2.new(1, -82, 0.5, -10),
            BackgroundColor3 = GOOD,
            BackgroundTransparency = 0.8,
            Font = Enum.Font.GothamBold,
            TextSize = 10,
            TextXAlignment = Enum.TextXAlignment.Center,
            TextColor3 = GOOD,
            Text = "OK",
        })
        corner(chip, 6)
        return { frame = f, name = n, chip = chip, txt = "OK", dname = plr.DisplayName, order = -1 }
    end

    local function updateList()
        local seen = {}
        local count = 0
        for i = 1, #playerList do
            local plr = playerList[i]
            seen[plr] = true
            count += 1
            local row = rows[plr]
            if not row then
                row = createRow(plr)
                rows[plr] = row
            end

            local tr = tracks[plr]
            local st = tr and tr.status or "OK"
            local col, txt = GOOD, "OK"
            if st == "SPIKE" then
                col, txt = BAD, "SPIKE x" .. tr.spikes
            elseif st == "FROZEN" then
                col, txt = WARN, "FROZEN"
            end

            if row.txt ~= txt then
                row.txt = txt
                row.chip.Text = txt
                row.chip.TextColor3 = col
                row.chip.BackgroundColor3 = col
            end
            if row.dname ~= plr.DisplayName then
                row.dname = plr.DisplayName
                row.name.Text = plr.DisplayName
            end
            local order = (st ~= "OK" and 0 or 1000) + i
            if row.order ~= order then
                row.order = order
                row.frame.LayoutOrder = order
            end
        end

        for plr, row in pairs(rows) do
            if not seen[plr] then
                row.frame:Destroy()
                rows[plr] = nil
            end
        end

        emptyLbl.Visible = (count == 0)
    end

    addUpdater(2, function()
        local flagged = state.flagged
        local c, txt
        if state.selfLagging then
            c, txt = WARN, "YOUR CONNECTION IS UNSTABLE\nDetection paused"
        elseif #flagged > 0 then
            local names = {}
            for i = 1, #flagged do
                local plr = flagged[i]
                local tr = tracks[plr]
                local n = plr.DisplayName
                if tr and tr.spikes > 0 then
                    n = n .. " x" .. tr.spikes
                end
                names[i] = n
            end
            c, txt = BAD, "LAG SWITCH DETECTED\n" .. table.concat(names, ", ")
        else
            c, txt = GOOD, "ALL CLEAR\nNo lag switchers detected"
        end
        bText.Text = txt
        bText.TextColor3 = c
        bStroke.Color = c
        updateList()
    end)
end

local function setMenuVisible(v)
    if v == menuOpen then return end
    menuOpen = v
    if v then
        Pill.Visible = false
        Root.Visible = true
        OpenScale.Scale = 0.92
        Main.GroupTransparency = 1
        tween(OpenScale, 0.25, { Scale = 1 }, Enum.EasingStyle.Back)
        tween(Main, 0.2, { GroupTransparency = 0 })
        task.delay(0.4, function()
            if menuOpen and not state.unloaded then
                OpenScale.Scale = 1
                Main.GroupTransparency = 0
            end
        end)
        runUpdaters()
    else
        tween(OpenScale, 0.15, { Scale = 0.92 })
        tween(Main, 0.15, { GroupTransparency = 1 })
        task.delay(0.16, function()
            if not menuOpen and not state.unloaded then
                Root.Visible = false
                Pill.Visible = true
            end
        end)
    end
end
